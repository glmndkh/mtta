];

export default function App() {
  // Группүүд: эхлээд A, B хоёр хоосон групп
  const [groups, setGroups] = useState([
    { id: 1, name: "A групп", players: [] },
    { id: 2, name: "B групп", players: [] },
  ]);

  // Тухайн групп дээр сонгосон тоглогчийн id хадгалах (select-ийн утга)
  const [selectedPlayerByGroup, setSelectedPlayerByGroup] = useState({});

  const totalGroups = groups.length;
  const totalAssignedPlayers = groups.reduce(
    (sum, g) => sum + g.players.length,
    0
  );

  const playersInAnyGroupIds = new Set(
    groups.flatMap((g) => g.players)
  );

  const getAvailablePlayersForGroup = () => {
    // Нэг тоглогч зөвхөн нэг группт байх логик – аль хэдийн хуваарилсан тоглогчдыг алгасна
    return registeredPlayers.filter((p) => !playersInAnyGroupIds.has(p.id));
  };

  const handleAddGroup = () => {
    setGroups((prev) => [
      ...prev,
      {
        id: Date.now(),
        name: `Шинэ групп ${prev.length + 1}`,
        players: [],
      },
    ]);
  };

  const handleSelectChange = (groupId, value) => {
    setSelectedPlayerByGroup((prev) => ({ ...prev, [groupId]: value }));
  };

  const handleAddPlayerToGroup = (groupId) => {
    const playerId = Number(selectedPlayerByGroup[groupId]);
    if (!playerId) return;

    setGroups((prev) =>
      prev.map((g) =>
        g.id === groupId && !g.players.includes(playerId)
          ? { ...g, players: [...g.players, playerId] }
          : g
      )
    );

    // тухайн группийн select-ийг цэвэрлэх
    setSelectedPlayerByGroup((prev) => ({ ...prev, [groupId]: "" }));
  };

  const getPlayerById = (id) =>
    registeredPlayers.find((p) => p.id === id);

  return (
    <div className="min-h-screen w-full bg-slate-100 flex flex-col items-center py-8">
      {/* Top header */}
      <header className="w-full max-w-6xl rounded-3xl bg-gradient-to-r from-indigo-500 to-purple-500 text-white px-8 py-6 shadow-lg flex flex-col gap-2">
        <div className="flex items-center gap-3 text-2xl font-semibold">
          <span className="text-3xl">🏆</span>
          <span>Тэмцээний үр дүн</span>
          <span className="text-3xl">🏆</span>
        </div>
        <p className="text-sm text-white/80">Админ хэсэг - Унших горим</p>

        {/* Tournament meta info */}
        <div className="mt-3 inline-flex items-center gap-3 rounded-2xl bg-white/10 px-4 py-2 text-xs backdrop-blur">
          <span className="font-semibold">UB Open 2025</span>
          <span className="h-4 w-px bg-white/40" />
          <span>Ганцаарчилсан тэмцээн</span>
          <span className="h-4 w-px bg-white/40" />
          <span>Нийт бүртгэлтэй тоглогч: {registeredPlayers.length}</span>
        </div>
      </header>

      {/* Main content */}
      <main className="w-full max-w-6xl mt-6 space-y-6">
        {/* Category card */}
        <section className="w-full rounded-3xl bg-white shadow-md border border-slate-200 px-8 py-6 space-y-6">
          {/* Title */}
          <div className="flex items-center gap-3">
            <div className="h-10 w-10 rounded-full bg-slate-100 flex items-center justify-center text-xl">🏅</div>
            <div className="flex flex-col">
              <span className="text-base font-semibold text-slate-800">Тэмцээний ангилал</span>
              <span className="text-xs text-slate-400">Тэмцээний төрөл, горимоо сонгоно уу</span>
            </div>
          </div>

          {/* Category tabs */}
          <div className="grid grid-cols-3 gap-3">
            <button className="flex items-center justify-center gap-2 rounded-2xl bg-blue-600 text-white py-3 text-sm font-medium shadow-sm">
              <span className="text-lg">👤</span>
              <span>Ганцаарчилсан тэмцээн</span>
            </button>
            <button className="flex items-center justify-center gap-2 rounded-2xl bg-slate-100 text-slate-700 py-3 text-sm font-medium border border-slate-200 hover:bg-slate-50 transition">
              <span className="text-lg">👥</span>
              <span>Хосын тэмцээн</span>
            </button>
            <button className="flex items-center justify-center gap-2 rounded-2xl bg-slate-100 text-slate-700 py-3 text-sm font-medium border border-slate-200 hover:bg-slate-50 transition">
              <span className="text-lg">🧑‍🤝‍🧑</span>
              <span>Багийн тэмцээн</span>
            </button>
          </div>

          {/* Divider */}
          <div className="h-px bg-slate-100" />

          {/* Division select + stage tabs */}
          <div className="space-y-4">
            {/* Division select */}
            <div className="space-y-1">
              <label className="text-xs font-medium text-slate-500 uppercase tracking-wide">
                Дивиз сонгох
              </label>
              <div className="relative w-64">
                <select className="w-full appearance-none rounded-xl border border-slate-200 bg-slate-50 px-3 py-2.5 text-sm text-slate-800 shadow-sm focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-blue-500">
                  <option>13–15 нас, эрэгтэй</option>
                  <option>13–15 нас, эмэгтэй</option>
                  <option>16–18 нас, эрэгтэй</option>
                </select>
                <div className="pointer-events-none absolute inset-y-0 right-3 flex items-center text-slate-400">
                  ▼
                </div>
              </div>
            </div>

            {/* Stage tabs */}
            <div className="flex gap-3 text-sm font-medium">
              <button className="flex-1 rounded-2xl bg-blue-600 text-white py-2.5 shadow-sm border border-blue-700">
                Групп шат
              </button>
              <button className="flex-1 rounded-2xl bg-white text-slate-700 py-2.5 border border-slate-300 hover:bg-slate-50 transition">
                Шууд хасагдах шат
              </button>
              <button className="flex-1 rounded-2xl bg-white text-slate-700 py-2.5 border border-slate-300 hover:bg-slate-50 transition">
                Эцсийн байр, медаль
              </button>
            </div>
          </div>
        </section>

        {/* Group stage section – групп үүсгээд, тоглогчдыг бүртгэлтэй жагсаалтаас сонгон хуваарилах */}
        <section className="w-full rounded-3xl bg-white shadow-md border border-slate-200 px-8 py-6 space-y-4">
          <div className="flex items-center justify-between gap-4">
            <div>
              <h2 className="text-base font-semibold text-slate-800">Групп шат (UB Open 2025)</h2>
              <p className="text-xs text-slate-400 mt-1">
                Эхлээд группүүдээ үүсгээд, дараа нь бүртгэлтэй тоглогчдоос сонгон
                групп тус бүрийн ширээ (table)-д хуваарилна.
              </p>
            </div>
            <button
              type="button"
              onClick={handleAddGroup}
              className="inline-flex items-center gap-2 rounded-2xl bg-blue-600 text-white text-sm font-medium px-4 py-2 shadow-sm hover:bg-blue-700 active:scale-[0.99] transition"
            >
              <span className="text-lg">＋</span>
              <span>Групп нэмэх</span>
            </button>
          </div>

          <div className="grid md:grid-cols-2 gap-4">
            {groups.map((group) => {
              const groupPlayers = group.players
                .map((id) => getPlayerById(id))
                .filter(Boolean);
              const availablePlayers = getAvailablePlayersForGroup();

              return (
                <div
                  key={group.id}
                  className="rounded-2xl border border-slate-200 bg-slate-50/70 px-4 py-4 flex flex-col gap-3"
                >
                  <div className="flex items-center justify-between gap-2">
                    <div className="flex items-center gap-2">
                      <span className="h-8 w-8 rounded-full bg-blue-100 text-blue-700 flex items-center justify-center text-sm font-semibold">
                        {group.name.charAt(0)}
                      </span>
                      <input
                        className="bg-transparent border-none outline-none text-sm font-semibold text-slate-800 px-0 py-0"
                        value={group.name}
                        onChange={(e) =>
                          setGroups((prev) =>
                            prev.map((g) =>
                              g.id === group.id
                                ? { ...g, name: e.target.value }
                                : g
                            )
                          )
                        }
                      />
                    </div>
                    <span className="text-[10px] uppercase tracking-wide text-slate-400">
                      Групп шат · Ганцаарчилсан
                    </span>
                  </div>

                  {/* Тоглогч нэмэх control */}
                  <div className="flex flex-col sm:flex-row gap-2 items-stretch sm:items-end text-xs">
                    <div className="flex-1 space-y-1">
                      <label className="text-[11px] text-slate-500">
                        Бүртгэлтэй тоглогч сонгох
                      </label>
                      <select
                        className="w-full rounded-xl border border-slate-200 bg-white px-3 py-1.5 text-xs text-slate-800 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
                        value={selectedPlayerByGroup[group.id] || ""}
                        onChange={(e) =>
                          handleSelectChange(group.id, e.target.value)
                        }
                      >
                        <option value="">— Сонгох —</option>
                        {availablePlayers.map((p) => (
                          <option key={p.id} value={p.id}>
                            {p.name} · {p.age} нас · {p.rating}
                          </option>
                        ))}
                      </select>
                    </div>
                    <button
                      type="button"
                      onClick={() => handleAddPlayerToGroup(group.id)}
                      disabled={availablePlayers.length === 0}
                      className="sm:w-32 rounded-xl bg-blue-600 text-white text-xs font-medium px-3 py-2 shadow-sm hover:bg-blue-700 disabled:bg-slate-300 disabled:cursor-not-allowed transition"
                    >
                      Тоглогч нэмэх
                    </button>
                  </div>

                  {/* Группийн тоглогчдын хүснэгт */}
                  <div className="mt-2 rounded-xl border border-slate-200 bg-white overflow-hidden">
                    <table className="w-full text-[11px]">
                      <thead className="bg-slate-100 text-slate-500">
                        <tr>
                          <th className="px-3 py-2 text-left w-10">#</th>
                          <th className="px-3 py-2 text-left">Нэр</th>
                          <th className="px-3 py-2 text-left w-16">Нас</th>
                          <th className="px-3 py-2 text-left w-16">Хүйс</th>
                          <th className="px-3 py-2 text-left w-20">Rating</th>
                        </tr>
                      </thead>
                      <tbody>
                        {groupPlayers.length === 0 ? (
                          <tr>
                            <td
                              colSpan={5}
                              className="px-3 py-3 text-center text-slate-400"
                            >
                              Энэ группт одоогоор тоглогч хуваарилаагүй байна.
                            </td>
                          </tr>
                        ) : (
                          groupPlayers.map((p, index) => (
                            <tr
                              key={p.id}
                              className={
                                index % 2 === 0 ? "bg-white" : "bg-slate-50"
                              }
                            >
                              <td className="px-3 py-1.5 text-slate-500">
                                {index + 1}
                              </td>
                              <td className="px-3 py-1.5 text-slate-800">
                                {p.name}
                              </td>
                              <td className="px-3 py-1.5 text-slate-700">
                                {p.age}
                              </td>
                              <td className="px-3 py-1.5 text-slate-700">
                                {p.gender}
                              </td>
                              <td className="px-3 py-1.5 text-slate-700">
                                {p.rating}
                              </td>
                            </tr>
                          ))
                        )}
                      </tbody>
                    </table>
                  </div>

                  <p className="text-[11px] text-slate-400">
                    * Нэг тоглогч зөвхөн нэг группт хуваарилагдана. Бүх бүртгэлтэй
                    тоглогчдыг хуваарилж дуусмагц сонголтын жагсаалт хоосорно.
                  </p>
                </div>
              );
            })}
          </div>
        </section>

        {/* Footer info line */}
        <section className="w-full max-w-6xl flex justify-between text-xs text-slate-500 px-2">
          <div className="space-x-4">
            <span>
              Нийт групп: <span className="font-semibold text-slate-700">{totalGroups}</span>
            </span>
            <span>
              Группт хуваарилсан тоглогч: <span className="font-semibold text-slate-700">{totalAssignedPlayers}</span>
            </span>
          </div>
          <div className="space-x-4">
            <span>Сүүлийн шинэчлэлт: 2025-11-17</span>
          </div>
        </section>
      </main>
    </div>
  );
}
